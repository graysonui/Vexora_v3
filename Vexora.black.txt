local Conns = {}
local function bind(sig, fn)
    local c = sig:Connect(fn)
    table.insert(Conns, c)
    return c
end

local function new(cls, props, parent)
    local inst = Instance.new(cls)
    for k, v in pairs(props or {}) do inst[k] = v end
    inst.Parent = parent
    return inst
end

local function round(p, r)
    return new("UICorner", { CornerRadius = UDim.new(0, r or 8) }, p)
end

local function stroke(p, col)
    return new("UIStroke", { Color = col or C.Border, Thickness = 1 }, p)
end

local function tw(obj, t, props, style, dir)
    local a = TweenService:Create(obj, TweenInfo.new(t or 0.25, style or Enum.EasingStyle.Quint, dir or Enum.EasingDirection.Out), props)
    a:Play()
    return a
end

local StarLayer = new("Frame", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, ZIndex = 1 }, Main)
local stars = {}
math.randomseed(os.clock() * 1000)
for i = 1, 22 do
    local sz = math.random(1, 2)
    local s = new("Frame", {
        Size = UDim2.fromOffset(sz, sz),
        Position = UDim2.fromScale(math.random(), math.random()),
        BackgroundColor3 = C.Accent,
        BackgroundTransparency = 0.6,
        BorderSizePixel = 0,
        ZIndex = 1,
    }, StarLayer)
    round(s, 2)
    table.insert(stars, { obj = s, sx = math.random(-10, 10)/1000, sy = math.random(-12, -3)/1000, ph = math.random()*6.28 })
end

bind(RunService.RenderStepped, function(dt)
    if not isOpen or not Main.Visible then return end
    local t = os.clock()
    for _, st in ipairs(stars) do
        local p = st.obj.Position
        st.obj.Position = UDim2.fromScale((p.X.Scale + st.sx * dt) % 1, (p.Y.Scale + st.sy * dt) % 1)
        st.obj.BackgroundTransparency = 0.55 + 0.35 * math.sin(t * 2.2 + st.ph)
    end
end)

local function makeDraggable(frame, handle, onClick)
    local dragging, moved, startPos, startInput
    bind(handle.InputBegan, function(inp)
        if inp.UserInputType == Enum.UserInputType.Touch or inp.UserInputType == Enum.UserInputType.MouseButton1 then
            dragging, moved, startInput, startPos = true, false, inp.Position, frame.Position
        end
    end)
    bind(UserInputService.InputChanged, function(inp)
        if dragging and (inp.UserInputType == Enum.UserInputType.Touch or inp.UserInputType == Enum.UserInputType.MouseMovement) then
            local delta = inp.Position - startInput
            if delta.Magnitude > 5 then moved = true end
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    bind(UserInputService.InputEnded, function(inp)
        if inp.UserInputType == Enum.UserInputType.Touch or inp.UserInputType == Enum.UserInputType.MouseButton1 then
            if dragging then dragging = false; if not moved and onClick then onClick() end end
        end
    end)
end

local function collapseUI()
    if isAnimating or not isOpen then return end
    isAnimating, isOpen = true, false
    tw(CloseBtn, 0.22, { Rotation = 90, BackgroundColor3 = C.Danger })
    tw(Main, 0.28, { Size = COLLAPSED_SIZE, GroupTransparency = 1 }, Enum.EasingStyle.Quint, Enum.EasingDirection.In)
    task.delay(0.24, function()
        Main.Visible, CloseBtn.Rotation, CloseBtn.BackgroundColor3 = false, 0, C.Card
        OpenPill.Visible, OpenPill.Size, OpenPill.GroupTransparency = true, UDim2.fromOffset(64, 22), 1
        tw(OpenPill, 0.26, { Size = UDim2.fromOffset(92, 28), GroupTransparency = 0 }, Enum.EasingStyle.Back)
        isAnimating = false
    end)
end

local function expandUI()
    if isAnimating or isOpen then return end
    isAnimating, isOpen = true, true
    tw(OpenPill, 0.16, { Size = UDim2.fromOffset(64, 22), GroupTransparency = 1 })
    task.delay(0.14, function()
        OpenPill.Visible, Main.Visible, Main.Size, Main.GroupTransparency = false, true, COLLAPSED_SIZE, 1
        tw(Main, 0.34, { Size = FULL_SIZE, GroupTransparency = 0 }, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
        task.delay(0.34, function() isAnimating = false end)
    end)
end

bind(CloseBtn.MouseButton1Click, collapseUI)

local Body = new("Frame", { Position = UDim2.fromOffset(8, 46), Size = UDim2.new(1, -16, 1, -54), BackgroundTransparency = 1, ZIndex = 2 }, Main)
local Pages, Tabs = {}, {}

local function switchTab(name)
    for k, page in pairs(Pages) do
        local active = (k == name)
        page.Visible = active
        local btn = Tabs[k]
        tw(btn, 0.18, { BackgroundColor3 = active and C.Card or C.Header, TextColor3 = active and C.Text or C.Muted })
        if active then
            tw(Indicator, 0.24, { Position = UDim2.new(0, btn.Position.X.Offset, 1, 3), Size = UDim2.fromOffset(btn.Size.X.Offset, 2) }, Enum.EasingStyle.Back)
        end
    end
end

local function createTopTab(name, width)
    local btn = new("TextButton", { Size = UDim2.fromOffset(width or 54, 28), BackgroundColor3 = C.Header, Text = name, Font = Enum.Font.GothamBold, TextSize = 10, TextColor3 = C.Muted, AutoButtonColor = false, ZIndex = 4 }, NavHost)
    round(btn, 6)
    local page = new("ScrollingFrame", { Size = UDim2.fromScale(1, 1), BackgroundTransparency = 1, BorderSizePixel = 0, ScrollBarThickness = 2, ScrollBarImageColor3 = C.Muted, CanvasSize = UDim2.new(), AutomaticCanvasSize = Enum.AutomaticSize.Y, Visible = false, ZIndex = 2 }, Body)
    new("UIListLayout", { Padding = UDim.new(0, 6), SortOrder = Enum.SortOrder.LayoutOrder }, page)
    Pages[name], Tabs[name] = page, btn
    bind(btn.MouseButton1Click, function() switchTab(name) end)
    return page
end
